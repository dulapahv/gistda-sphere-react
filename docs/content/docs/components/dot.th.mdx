```tsx
import { Dot, SphereMap } from "gistda-sphere-react";

<SphereMap center={{ lon: 100.5018, lat: 13.7563 }} style={{ height: 400 }} zoom={12}>
  <Dot position={{ lon: 100.5018, lat: 13.7563 }} lineColor="red" lineWidth={12} title="Bangkok" />
  <Dot position={{ lon: 100.52, lat: 13.77 }} lineColor="blue" lineWidth={8} title="Point B" />
</SphereMap>
```

ต่างจาก [`Circle`](/th/docs/components/circle) ที่รัศมีเป็นระยะทางบนพื้นโลก จุดจะมีขนาดบนหน้าจอเท่าเดิมทุกระดับซูม โดย `lineWidth` คือเส้นผ่านศูนย์กลางเป็นพิกเซล

## Props

| Prop | ชนิด | คำอธิบาย |
| --- | --- | --- |
| `position` | `Location` | **จำเป็น** ตำแหน่งของจุด เมื่อเปลี่ยนค่า จุดจะย้ายโดยไม่ต้องสร้างใหม่ |
| `onDrop` | `(dot, location) => void` | วางจุดที่ลากได้ลงที่ `location` |

จุดรับ [props ของรูปทรง](/th/docs/components/polygon#shape-props) ด้วย เช่น `lineColor`, `lineWidth`, `draggable` และ `title`

## Ref

| เมธอด | คำอธิบาย |
| --- | --- |
| `getDot()` | `sphere.Dot` ที่อยู่เบื้องหลัง |
| `getPosition()` / `setPosition(location)` | อ่านหรือย้ายตำแหน่งจุด |
| `contains`, `intersects`, `within`, `distanceTo`, `updateStyle`, `togglePopup`, `toGeoJSON` | เหมือนกับ[รูปหลายเหลี่ยม](/th/docs/components/polygon#ref) |

`setPosition` และการลากจะย้ายจุดจนกว่า prop `position` จะเปลี่ยนเป็นตำแหน่งใหม่ เช่นเดียวกับ[หมุด](/th/docs/components/marker#ref-or-position) หากต้องการให้จุดอยู่ในตำแหน่งที่ย้ายไป ให้เก็บตำแหน่งไว้ใน state เช่น จาก `onDrop`
