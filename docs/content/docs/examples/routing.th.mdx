วางแผนเส้นทางระหว่างสองเมือง แล้วแสดงคำแนะนำการเดินทาง

<RouteDemo />

```tsx lineNumbers
import { RouteGuide, SphereMap, SphereProvider, useRoute, type RouteMode } from "gistda-sphere-react";
import { useState } from "react";

const BANGKOK = { lon: 100.5018, lat: 13.7563 };
const CHIANG_MAI = { lon: 98.9853, lat: 18.7883 };

function Directions() {
  const route = useRoute();
  const [mode, setMode] = useState<RouteMode>("Traffic");

  const plan = () => {
    route.clear();
    route.setMode(mode);
    route.addDestination(BANGKOK);
    route.addDestination(CHIANG_MAI);
    route.search();
  };

  return (
    <div>
      <select value={mode} onChange={(event) => setMode(event.target.value as RouteMode)}>
        <option value="Traffic">Avoid traffic</option>
        <option value="Distance">Shortest</option>
        <option value="Fly">Straight line</option>
      </select>
      <button onClick={plan} disabled={!route.isReady || route.isCalculating}>
        {route.isCalculating ? "Calculating…" : "Plan route"}
      </button>

      {route.error && <p role="alert">{route.error.message}</p>}

      {route.summary && (
        <p>
          {route.summary.distanceText} · {route.summary.intervalText}
        </p>
      )}

      <ol>
        {route.steps.map((step, index) => (
          <li key={index}>
            {step.instruction} <small>{step.distanceText}</small>
          </li>
        ))}
      </ol>
    </div>
  );
}

function App() {
  return (
    <SphereProvider apiKey="YOUR_API_KEY">
      <Directions />
      <SphereMap style={{ height: 500 }} />
    </SphereProvider>
  );
}
```

## รูปแบบอื่น [#variations]

- **หลายจุดแวะ**: เรียก `addDestination` สำหรับทุกจุด จุดที่ต่อกันแต่ละคู่คือหนึ่งช่วงใน `route.routes`
- **ให้ผู้ใช้วางแผนบนแผนที่**: `route.enableContextMenu()` เพิ่มเมนู "เดินทางมาจุดนี้" ในเมนูคลิกขวา ผู้ใช้ลากจุดแวะได้ และเส้นทางจะอัปเดตเอง
- **แผงคำแนะนำของ Sphere**: แสดง [`<RouteGuide>`](/th/docs/components/route-guide) ข้างแผนที่
- **ไม่ใช้แผนที่เลย**: [บริการเว็บการนำทาง](/th/docs/web-services/routing)คืนข้อมูลชุดเดียวกันในรูปแบบ GeoJSON

<Callout>
  หากไม่มีอะไรเกิดขึ้นและได้รับข้อผิดพลาด `"timeout"` หลังผ่านไป 20 วินาที แสดงว่า API key ของคุณยังไม่ได้เปิดบริการนำทาง ส่วนโหมด "เส้นตรง" ใช้งานได้โดยไม่ต้องเปิดบริการนี้
</Callout>
