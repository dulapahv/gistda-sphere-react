## ใช้ hooks แทน [#with-hooks-instead]

ตัวอย่างนี้เก็บรูปทรงไว้ใน state ของคอมโพเนนต์ หากต้องการเก็บไว้ใน hook แทน และรับการคลิกจากคอมโพเนนต์ลูกของแผนที่ ให้ใช้ [`usePolygons`](/th/docs/hooks/overlay-hooks) และ [`useMapEvent`](/th/docs/hooks/event-hooks):

```tsx lineNumbers
import { useState } from "react";
import {
  Dot, Polygon, SphereMap, SphereProvider,
  useMapEvent, usePolygons, type Location,
} from "gistda-sphere-react";

/** Drops repeated points: the two clicks of a double-click land on the same spot. */
const withoutRepeats = (points: Location[]) =>
  points.filter(
    (point, index) =>
      index === 0 || point.lat !== points[index - 1].lat || point.lon !== points[index - 1].lon
  );

function PolygonDrawer() {
  const [points, setPoints] = useState<Location[]>([]);
  const { items: polygons, add } = usePolygons();

  useMapEvent("click", ({ lat, lon }) => {
    setPoints((previous) => [...previous, { lat, lon }]);
  });

  useMapEvent("doubleClick", () => {
    const corners = withoutRepeats(points);
    if (corners.length >= 3) {
      add({ positions: corners, fillColor: "rgba(255,0,0,0.3)", lineColor: "red" });
      setPoints([]);
    }
  });

  return (
    <>
      {polygons.map(({ id, ...polygon }) => (
        <Polygon key={id} {...polygon} />
      ))}
      {points.map((point, index) => (
        <Dot key={index} position={point} lineColor="red" lineWidth={8} />
      ))}
    </>
  );
}

function App() {
  return (
    <SphereProvider apiKey="YOUR_API_KEY">
      <SphereMap
        center={{ lon: 100.5, lat: 13.75 }}
        zoom={10}
        style={{ height: 500 }}
        options={{ doubleClickZoom: false }}
      >
        <PolygonDrawer />
      </SphereMap>
    </SphereProvider>
  );
}
```

## เคล็ดลับ [#tips]

- การดับเบิลคลิกจะส่งการคลิกสองครั้งมาด้วย ให้ตัดจุดที่ซ้ำกันออกก่อนสร้างรูปทรง ตามตัวอย่างด้านบน
- `options={{ doubleClickZoom: false }}` ทำให้ดับเบิลคลิกไม่ซูมแผนที่ หากต้องการให้ซูมได้ตามปกติในเวลาอื่น ให้ส่งค่านี้เฉพาะตอนที่กำลังวาด
- Sphere มีแถบเครื่องมือวาดในตัวด้วย เปิดได้ด้วย `controls={{ Toolbar: true }}` และรับรูปทรงที่วาดจาก `onDrawCreate`
- หน้า[ทดลองเล่น](/th/playground)มีเครื่องมือวาดครบชุด ทั้งเส้น วงกลม สี่เหลี่ยม การวัดขนาด และการวิเคราะห์เชิงพื้นที่
