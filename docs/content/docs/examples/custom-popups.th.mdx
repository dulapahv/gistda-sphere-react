## ไอคอนหมุดแบบ React [#react-marker-icons]

```tsx lineNumbers
import { type Location, Marker } from "gistda-sphere-react";
import { useState } from "react";

interface Hotel {
  id: string;
  location: Location;
  name: string;
  photo: string;
  price: number;
  rating: number;
  reviews: number;
}

function PriceMarker({ hotel }: { hotel: Hotel }) {
  const [hovered, setHovered] = useState(false);
  return (
    <Marker
      position={hotel.location}
      title={hotel.name}
      onHover={() => setHovered(true)}
      onLeave={() => setHovered(false)}
      options={{ anchor: "bottom" }}
    >
      <span className={hovered ? "price price--active" : "price"}>฿ {hotel.price}</span>
    </Marker>
  );
}
```

## ป๊อปอัปแบบ React [#react-popups]

```tsx lineNumbers
import { Marker } from "gistda-sphere-react";

<Marker
  position={hotel.location}
  title={hotel.name}
  popupContent={
    <div className="w-56">
      <img src={hotel.photo} alt="" className="rounded" />
      <p>{hotel.rating} ★ · {hotel.reviews} reviews</p>
      <button onClick={() => book(hotel.id)}>Book</button>
    </div>
  }
/>
```

รูปทรงต่าง ๆ ก็ใช้ `popupContent` ได้เช่นกัน

```tsx
import { Polygon } from "gistda-sphere-react";

<Polygon positions={district.outline} title={district.name} popupContent={<DistrictStats id={district.id} />} />
```

## ป๊อปอัปของ POI ของ Sphere [#popups-for-spheres-own-pois]

ชั้นข้อมูลแท็กและผลการค้นหาจะเปิดป๊อปอัปของ Sphere สามารถปรับเปลี่ยนเนื้อหาที่แสดงได้ที่ `SphereProvider`:

```tsx lineNumbers
import { SphereProvider } from "gistda-sphere-react";

<SphereProvider
  apiKey={apiKey}
  formatPoi={(poi) => {
    const element = document.createElement("div");
    const name = document.createElement("strong");
    name.textContent = poi.name;
    const address = document.createElement("p");
    address.textContent = poi.address ?? "";
    element.append(name, address);
    return element;
  }}
>
```

หากต้องการควบคุมทั้งหมด ให้คืนค่าตัวเลือกป๊อปอัปจาก `poiPopupOptions(source, id, name, language)` เช่น เพื่อโหลดรายละเอียดจาก API ของคุณเอง ดู [SphereProvider](/th/docs/components/sphere-provider#customizing-poi-popups)
