เลือกหมวดหมู่ แล้วเปิดหรือปิดแท็กเพื่อแสดงสถานที่เหล่านั้นบนแผนที่

`<Tag>` แต่ละตัวจะเพิ่มชั้นข้อมูลเมื่อ mount และลบออกเมื่อ unmount แผนที่จึงตรงกับค่า `activeTags` เสมอ หากต้องการจัดการแท็กแบบ imperative แทน ให้ใช้ [`useTags`](/th/docs/hooks/use-tags):

```tsx
import { useTags } from "gistda-sphere-react";

const { add, remove, clear } = useTags();
add("โรงพยาบาล");
add("เอทีเอ็ม");
remove("โรงพยาบาล");
```
