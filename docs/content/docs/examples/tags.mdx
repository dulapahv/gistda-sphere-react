title: Points of interest by category
description: Let users pick POI categories and show them on the map
---

Pick a category, then toggle tags to show those places on the map:

<TagsDemo />

```tsx lineNumbers
import { SphereMap, SphereProvider, Tag, useTagCategories } from "gistda-sphere-react";
import { useState } from "react";

function TagPicker() {
  const { categories } = useTagCategories("en");
  const [active, setActive] = useState<string[]>([]);

  const toggle = (id: string) =>
    setActive((previous) =>
      previous.includes(id) ? previous.filter((tag) => tag !== id) : [...previous, id]
    );

  return (
    <>
      {categories.map((category) => (
        <details key={category.name}>
          <summary>{category.name}</summary>
          {category.tags.map((tag) => (
            <button key={tag.id} aria-pressed={active.includes(tag.id)} onClick={() => toggle(tag.id)}>
              {tag.label}
            </button>
          ))}
        </details>
      ))}

      <SphereMap style={{ height: 500 }} zoom={12}>
        {active.map((tag) => (
          <Tag key={tag} name={tag} />
        ))}
      </SphereMap>
    </>
  );
}

function App() {
  return (
    <SphereProvider apiKey="YOUR_API_KEY">
      <TagPicker />
    </SphereProvider>
  );
}
```

Each `<Tag>` adds its layer when it mounts and removes it when it unmounts, so the map always matches `active`. To manage tags imperatively instead, use [`useTags`](/docs/hooks/use-tags):

```tsx
const { add, remove, clear } = useTags();
add("โรงพยาบาล");
add("เอทีเอ็ม");
remove("โรงพยาบาล");
```
