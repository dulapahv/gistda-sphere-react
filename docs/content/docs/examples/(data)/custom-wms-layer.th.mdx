```tsx lineNumbers
import { Layer, SphereMap, SphereProvider } from "gistda-sphere-react";

function App() {
  return (
    <SphereProvider apiKey="YOUR_API_KEY">
      <SphereMap center={{ lon: 101, lat: 13 }} zoom={5} style={{ height: 500 }}>
        <Layer
          name="bluemarble_terrain"
          type="WMS"
          url="https://ms.longdo.com/mapproxy/service"
          opacity={0.7}
          zoomRange={{ min: 1, max: 12 }}
        />
      </SphereMap>
    </SphereProvider>
  );
}
```

`name` คือชื่อชั้นข้อมูลของ WMS (พารามิเตอร์ `LAYERS`) หากใช้เซิร์ฟเวอร์อื่น ให้ตรวจสอบค่าเหล่านี้เพิ่มเติม:

| Prop | ควรกำหนดเมื่อ |
| --- | --- |
| `version` | เซิร์ฟเวอร์รองรับเฉพาะ WMS `1.3.0` (ค่าเริ่มต้น `1.1.1`) |
| `srs` | เซิร์ฟเวอร์ไม่มี `EPSG:3857` |
| `format` | ต้องการ `image/jpeg` หรือรูปแบบอื่นแทน `image/png` |
| `styles` | ชั้นข้อมูลมีสไตล์ที่ระบุชื่อไว้ |
| `extraQuery` | เซิร์ฟเวอร์ต้องการพารามิเตอร์เพิ่ม เช่น `TIME=2024-01-01` หรือ `TRANSPARENT=true` |

ใช้วิธีเดียวกันกับบริการ WMTS, TMS และ XYZ ได้ ดู [`Layer`](/th/docs/components/layer#tile-services)
