Plan a route between two cities and list the directions:

<RouteDemo />

```tsx lineNumbers
import { RouteGuide, SphereMap, SphereProvider, useRoute, type RouteMode } from "gistda-sphere-react";
import { useState } from "react";

const BANGKOK = { lon: 100.5018, lat: 13.7563 };
const CHIANG_MAI = { lon: 98.9853, lat: 18.7883 };

function Directions() {
  const route = useRoute();
  const [mode, setMode] = useState<RouteMode>("Traffic");

  const plan = () => {
    route.clear();
    route.setMode(mode);
    route.addDestination(BANGKOK);
    route.addDestination(CHIANG_MAI);
    route.search();
  };

  return (
    <div>
      <select value={mode} onChange={(event) => setMode(event.target.value as RouteMode)}>
        <option value="Traffic">Avoid traffic</option>
        <option value="Distance">Shortest</option>
        <option value="Fly">Straight line</option>
      </select>
      <button onClick={plan} disabled={!route.isReady || route.isCalculating}>
        {route.isCalculating ? "Calculating…" : "Plan route"}
      </button>

      {route.error && <p role="alert">{route.error.message}</p>}

      {route.summary && (
        <p>
          {route.summary.distanceText} · {route.summary.intervalText}
        </p>
      )}

      <ol>
        {route.steps.map((step, index) => (
          <li key={index}>
            {step.instruction} <small>{step.distanceText}</small>
          </li>
        ))}
      </ol>
    </div>
  );
}

function App() {
  return (
    <SphereProvider apiKey="YOUR_API_KEY">
      <Directions />
      <SphereMap style={{ height: 500 }} />
    </SphereProvider>
  );
}
```

## Variations

- **Several stops**: call `addDestination` for each stop; each pair of stops is a leg in `route.routes`.
- **Let users plan on the map**: `route.enableContextMenu()` adds "Direction to here" to the right-click menu. Stops can be dragged, and the route updates by itself.
- **Sphere's own panel**: render [`<RouteGuide>`](/docs/components/route-guide) next to the map.
- **No map at all**: the [routing web service](/docs/web-services/routing) returns the same data as GeoJSON.

<Callout>
  If nothing happens and you get a `"timeout"` error after 20 seconds, your API key doesn't have the routing service. "Straight line" mode works without it.
</Callout>
