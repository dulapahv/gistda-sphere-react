`useMapControls` ควบคุมแผนที่แบบ imperative เช่น จากปุ่มในแถบด้านข้าง ใช้งานได้ทุกที่ภายใต้ `SphereProvider` และมีเมธอดชุดเดียวกับ [ref ของ `SphereMap`](/th/docs/components/sphere-map#ref)

```tsx
import { useMapControls } from "gistda-sphere-react";

function Toolbar() {
  const { goTo, zoomIn, zoomOut, setRotate, setPitch } = useMapControls();
  return (
    <>
      <button onClick={() => goTo({ center: { lon: 98.9853, lat: 18.7883 }, zoom: 12 })}>Chiang Mai</button>
      <button onClick={() => zoomIn()}>+</button>
      <button onClick={() => zoomOut()}>−</button>
      <button onClick={() => { setRotate(0); setPitch(0); }}>Reset view</button>
    </>
  );
}
```

## มุมมอง [#camera]

| เมธอด | คำอธิบาย |
| --- | --- |
| `goTo(options, animate?)` | เคลื่อนไปยังมุมมองที่กำหนด `options`: `center`, `zoom`, `rotate` (หรือ `bearing`), `pitch`, `padding`, `duration` โดยค่าเริ่มต้นจะมีแอนิเมชัน |
| `getView()` | จุดกึ่งกลาง ระดับซูม มุมหมุน มุมเอียง และขอบเขตที่มองเห็นในอ็อบเจกต์เดียว หรือ `null` หากแผนที่ยังไม่พร้อมใช้งาน |
| `setCenter(location, animate?)` / `getCenter()` | จุดกึ่งกลางของแผนที่ |
| `setZoom(zoom, animate?)` / `getZoom()` | ระดับซูม |
| `zoomIn(animate?)` / `zoomOut(animate?)` | ซูมเข้าหรือออกหนึ่งระดับ |
| `setRotate(angle, animate?)` / `getRotate()` | มุมหมุนเป็นองศา ตามเข็มนาฬิกาจากทิศเหนือ |
| `setPitch(angle)` / `getPitch()` | มุมเอียงเป็นองศา |
| `setBound(bound, options?)` / `getBound()` | ปรับมุมมองให้พอดีกับกรอบ `{ minLon, minLat, maxLon, maxLat }` (พร้อม `padding`, `maxZoom`…) หรืออ่านกรอบที่มองเห็น |
| `panBy({ x, y }, animate?)` | เลื่อนแผนที่ตามจำนวนพิกเซล |
| `setZoomRange(range)` / `getZoomRange()` | ช่วงระดับซูมที่ผู้ใช้ซูมได้ |

`rotate` และ `bearing` คือมุมเดียวกัน โดย `rotate` ตรงกับ prop ของ `SphereMap` และ `setRotate` ส่วน `bearing` เป็นชื่อที่ MapLibre ใช้ หากส่งทั้งสองค่า จะใช้ค่า `bearing`

## ชั้นข้อมูลและการแสดงผล [#layers-and-appearance]

| เมธอด | คำอธิบาย |
| --- | --- |
| `setBaseLayer(layer)` | เปลี่ยนแผนที่ฐาน: ชื่อชั้นข้อมูลในตัว, `sphere.Layer` หรือ URL/อ็อบเจกต์สไตล์ของ MapLibre |
| `addLayer(layer, beforeId?)` / `removeLayer(layer)` | เพิ่มหรือลบชั้นข้อมูลที่ซ้อนอยู่บนแผนที่ฐาน |
| `clearLayers()` / `getLayers()` | ลบชั้นข้อมูลที่เพิ่มไว้ทั้งหมด / ดูรายการชั้นข้อมูลของแผนที่ |
| `setFilter(filter)` / `getFilter()` | ตัวกรองสี: `"Dark"`, `"Light"`, `"Protanopia"` หรือ `"Deuteranopia"` ส่ง `"None"` หรือ `false` เพื่อล้างตัวกรอง |
| `setLanguage(language)` / `getLanguage()` | ภาษาของป้ายชื่อและ UI `"th"` หรือ `"en"` |
| `loadPredefinedOverlay(type, options?)` / `unloadPredefinedOverlay(type)` | ข้อมูลสด `"cameras"`, `"events"` หรือ `"aqi"` |

## อื่น ๆ [#other]

| เมธอด | คำอธิบาย |
| --- | --- |
| `isReady` | เป็น `true` เมื่อแผนที่พร้อมใช้งาน ก่อนหน้านั้นเมธอดต่าง ๆ จะไม่ทำงาน |
| `resize()` | วัดขนาดแผนที่ใหม่หลังคอนเทนเนอร์เปลี่ยนขนาด |
| `repaint()` | วาดแผนที่ใหม่ |

<Callout>
  สำหรับสถานะที่ควรเปลี่ยนตามแอป (ภาษา ตัวกรองสี แผนที่ฐาน ขอบเขตการซูม ตัวควบคุม) การใช้ [props ของ `SphereMap`](/th/docs/components/sphere-map#props) จะง่ายกว่า ส่วนเมธอดเหล่านี้เหมาะกับการสั่งงาน เช่น บินไปยังสถานที่เมื่อคลิกรายการ
</Callout>
